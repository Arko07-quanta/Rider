  reviews,
  ride_cancellations,
  transactions,

DROP TYPE IF EXISTS
  discount_type,
  review_type,
  cancel_actor,
  transaction_status,
  transaction_type,
  ride_status,
  ride_request_status,
  user_status,
  user_role;

CREATE TYPE user_role AS ENUM ('admin','rider','driver');
CREATE TYPE user_status AS ENUM ('active','inactive','suspended');
CREATE TYPE ride_request_status AS ENUM ('pending','accepted','rejected','cancelled');
CREATE TYPE ride_status AS ENUM ('ongoing','completed','cancelled','no_show');
CREATE TYPE transaction_type AS ENUM ('credit','debit');
CREATE TYPE transaction_status AS ENUM ('pending','completed','failed');
CREATE TYPE cancel_actor AS ENUM ('rider','driver','admin');
CREATE TYPE review_type AS ENUM ('rider_to_driver','driver_to_rider');
CREATE TYPE discount_type AS ENUM ('percentage','fixed');

CREATE TABLE users (
  user_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  email VARCHAR(100) UNIQUE NOT NULL,
  phone VARCHAR(20) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  role user_role NOT NULL,
  token_version INT DEFAULT 1,
  status user_status DEFAULT 'active',
  email_verified BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP
);

CREATE TABLE admins (
  user_id BIGINT PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
  access_level INT NOT NULL
);

CREATE TABLE riders (
  user_id BIGINT PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE
);

CREATE TABLE drivers (
  user_id BIGINT PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
  license_number VARCHAR(50) UNIQUE NOT NULL,
  rating NUMERIC(3,2) DEFAULT 5.00 CHECK (rating BETWEEN 0 AND 5),
  is_verified BOOLEAN DEFAULT FALSE,
  current_lat NUMERIC(9,6),
  current_lng NUMERIC(9,6),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP
);

CREATE TABLE locations (
  location_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  latitude NUMERIC(9,6) NOT NULL,
  longitude NUMERIC(9,6) NOT NULL,
  address TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE vehicle_types (
  vehicle_type_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  type_name VARCHAR(50) NOT NULL,
  max_passengers INT NOT NULL,
  max_weight NUMERIC(10,2),
  average_speed NUMERIC(6,2),
  base_fare NUMERIC(10,2) NOT NULL,
  fare_per_km NUMERIC(10,2) NOT NULL,
  fare_per_minute NUMERIC(10,2) NOT NULL,
  minimum_fare NUMERIC(10,2) NOT NULL,
  cancellation_fare NUMERIC(10,2) DEFAULT 0.00
);

CREATE TABLE vehicles (
  vehicle_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  driver_id BIGINT REFERENCES drivers(user_id) ON DELETE SET NULL,
  vehicle_type_id BIGINT NOT NULL REFERENCES vehicle_types(vehicle_type_id),
  plate_number VARCHAR(20) UNIQUE NOT NULL,
  brand VARCHAR(50),
  model VARCHAR(50),
  year INT,
  color VARCHAR(30),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP
);

CREATE TABLE wallets (
  wallet_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  user_id BIGINT UNIQUE NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  balance NUMERIC(15,2) DEFAULT 0.00,
  currency CHAR(3) DEFAULT 'USD',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP
);

CREATE TABLE payment_types (
  payment_type_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  company_name VARCHAR(100) NOT NULL,
  logo VARCHAR(255),
  description TEXT
);

CREATE TABLE payment_accounts (
  payment_acc_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  user_id BIGINT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  payment_type_id BIGINT NOT NULL REFERENCES payment_types(payment_type_id),
  account_identifier VARCHAR(255) NOT NULL,
  metadata JSONB,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE saved_locations (
  saved_location_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  rider_id BIGINT NOT NULL REFERENCES riders(user_id) ON DELETE CASCADE,
  location_id BIGINT NOT NULL REFERENCES locations(location_id) ON DELETE CASCADE,
  label VARCHAR(50),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE ride_requests (
  request_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  rider_id BIGINT NOT NULL REFERENCES riders(user_id) ON DELETE CASCADE,
  vehicle_type_id BIGINT NOT NULL REFERENCES vehicle_types(vehicle_type_id),
  requested_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  status ride_request_status DEFAULT 'pending',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE rides (
  ride_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  request_id BIGINT UNIQUE NOT NULL REFERENCES ride_requests(request_id),
  driver_id BIGINT REFERENCES drivers(user_id) ON DELETE SET NULL,
  vehicle_id BIGINT REFERENCES vehicles(vehicle_id) ON DELETE SET NULL,
  pickup_location_id BIGINT NOT NULL REFERENCES locations(location_id),
  dropoff_location_id BIGINT NOT NULL REFERENCES locations(location_id),
  start_time TIMESTAMP,
  end_time TIMESTAMP,
  distance NUMERIC(10,2) DEFAULT 0.00 CHECK (distance >= 0),
  duration NUMERIC(10,2) DEFAULT 0.00 CHECK (duration >= 0),
  fare NUMERIC(10,2) DEFAULT 0.00 CHECK (fare >= 0),
  status ride_status DEFAULT 'ongoing',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP
);

CREATE TABLE transactions (
  transaction_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  wallet_id BIGINT NOT NULL REFERENCES wallets(wallet_id) ON DELETE CASCADE,
  ride_id BIGINT REFERENCES rides(ride_id) ON DELETE SET NULL,
  type transaction_type NOT NULL,
  amount NUMERIC(15,2) NOT NULL CHECK (amount >= 0),
  timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  status transaction_status DEFAULT 'completed',
  payment_method VARCHAR(50),
  external_ref VARCHAR(128)
);

CREATE TABLE ride_cancellations (
  cancellation_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  ride_id BIGINT UNIQUE NOT NULL REFERENCES rides(ride_id) ON DELETE CASCADE,
  cancelled_by cancel_actor NOT NULL,
  reason TEXT,
  timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE reviews (
  review_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  ride_id BIGINT NOT NULL REFERENCES rides(ride_id) ON DELETE CASCADE,
  author_id BIGINT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  rating SMALLINT NOT NULL CHECK (rating BETWEEN 1 AND 5),
  comment TEXT,
  review_type review_type NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE chat_messages (
  message_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  ride_id BIGINT NOT NULL REFERENCES rides(ride_id) ON DELETE CASCADE,
  sender_id BIGINT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  message_text TEXT NOT NULL,
  sent_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  is_read BOOLEAN DEFAULT FALSE
);

CREATE TABLE trip_history (
  history_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  ride_id BIGINT NOT NULL REFERENCES rides(ride_id) ON DELETE CASCADE,
  location_id BIGINT REFERENCES locations(location_id) ON DELETE SET NULL,
  latitude NUMERIC(9,6),
  longitude NUMERIC(9,6),
  timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE driver_status (
  driver_id BIGINT PRIMARY KEY REFERENCES drivers(user_id) ON DELETE CASCADE,
  is_online BOOLEAN DEFAULT FALSE,
  is_available BOOLEAN DEFAULT FALSE,
  last_seen TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  latitude NUMERIC(9,6),
  longitude NUMERIC(9,6)
);

CREATE TABLE promotions (
  promo_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  code VARCHAR(50) UNIQUE NOT NULL,
  discount_type discount_type NOT NULL,
  value NUMERIC(10,2) NOT NULL,
  discount_amount NUMERIC(10,2),
  usage_limit INT,
  expiry_date DATE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE user_coupons (
  user_coupon_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  user_id BIGINT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  promo_id BIGINT NOT NULL REFERENCES promotions(promo_id) ON DELETE CASCADE,
  is_used BOOLEAN DEFAULT FALSE,
  earned_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(user_id, promo_id)
);

-- ─── TRIGGER FUNCTIONS ───

CREATE OR REPLACE FUNCTION fn_reward_rider_coupons()
RETURNS TRIGGER AS $$
DECLARE
    v_rider_id BIGINT;
    v_ride_count INT;
    v_promo_id BIGINT;
BEGIN
    -- Only trigger when a ride is COMPLETED
    IF (NEW.status = 'completed') THEN
        -- Get the rider_id from the ride_requests table
        SELECT rider_id INTO v_rider_id 
        FROM ride_requests 
        WHERE request_id = NEW.request_id;

        -- Count successfully completed rides for this rider
        SELECT COUNT(*) INTO v_ride_count 
        FROM rides r
        JOIN ride_requests req ON r.request_id = req.request_id
        WHERE req.rider_id = v_rider_id AND r.status = 'completed';

        -- 1. Reward on 1st Ride (Welcome Coupon)
        IF v_ride_count = 1 THEN
            SELECT promo_id INTO v_promo_id FROM promotions WHERE code = 'WELCOME20' LIMIT 1;
            IF v_promo_id IS NOT NULL THEN
                INSERT INTO user_coupons (user_id, promo_id) 
                VALUES (v_rider_id, v_promo_id)
                ON CONFLICT DO NOTHING;
            END IF;
        END IF;

        -- 2. Reward on 20th Ride (Loyalty Coupon)
        IF v_ride_count = 20 THEN
            SELECT promo_id INTO v_promo_id FROM promotions WHERE code = 'LOYALTY50' LIMIT 1;
            IF v_promo_id IS NOT NULL THEN
                INSERT INTO user_coupons (user_id, promo_id) 
                VALUES (v_rider_id, v_promo_id)
                ON CONFLICT DO NOTHING;
            END IF;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ─── TRIGGERS ───

CREATE TRIGGER trg_after_ride_complete
AFTER UPDATE OF status ON rides
FOR EACH ROW
EXECUTE FUNCTION fn_reward_rider_coupons();

-- ─── STORED PROCEDURES ───

CREATE OR REPLACE PROCEDURE sp_system_maintenance(p_admin_id BIGINT)
LANGUAGE plpgsql
AS $$
BEGIN
    -- 1. Log the start of maintenance
    INSERT INTO admin_actions (admin_id, action_description)
    VALUES (p_admin_id, 'SYSTEM_MAINTENANCE: Cleaning up logs and optimizing system tables...');

    -- 2. Cleanup old chat messages (older than 30 days)
    DELETE FROM chat_messages WHERE sent_at < (CURRENT_TIMESTAMP - INTERVAL '30 days');

    -- 3. Cleanup very old admin actions (older than 90 days)
    DELETE FROM admin_actions WHERE timestamp < (CURRENT_TIMESTAMP - INTERVAL '90 days');

    -- 4. Mark the end of maintenance
    INSERT INTO admin_actions (admin_id, action_description)
    VALUES (p_admin_id, 'SYSTEM_MAINTENANCE: Completed successfully.');
    
    COMMIT;
END;
$$;

CREATE TABLE admin_actions (
  action_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  admin_id BIGINT REFERENCES admins(user_id) ON DELETE SET NULL,
  action_description TEXT,
  timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
